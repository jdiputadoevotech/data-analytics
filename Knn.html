# Load and clean data [cite: 1248, 1251]
data("titanic_train")
titanic <- titanic_train[, c("Survived", "Pclass", "Sex", "Age", "Fare")]
titanic <- na.omit(titanic)

# Factorize categorical variables [cite: 1269]
titanic$Survived <- factor(titanic$Survived, levels = c(0, 1), labels = c("Died", "Survived"))
titanic$Sex <- factor(titanic$Sex)

# Split data into training (70%) and testing (30%) sets [cite: 54, 57]
set.seed(123) 
train_index <- createDataPartition(titanic$Survived, p = 0.7, list = FALSE)
train_data <- titanic[train_index, ]
test_data  <- titanic[-train_index, ]

# Create copies for scaling [cite: 64]
train_scaled <- train_data
test_scaled <- test_data

# Scale numeric features 
num_cols <- c("Pclass", "Age", "Fare")
preProc <- preProcess(train_data[, num_cols], method = c("center", "scale"))
train_scaled[, num_cols] <- predict(preProc, train_data[, num_cols])
test_scaled[, num_cols] <- predict(preProc, test_data[, num_cols])

# Convert Sex to numeric for distance calculation
train_scaled$Sex <- as.numeric(train_scaled$Sex)
test_scaled$Sex <- as.numeric(test_scaled$Sex)

# Apply kNN with k=3 [cite: 71, 72]
knn_result <- knn(train = train_scaled[, -1], 
                  test = test_scaled[, -1], 
                  cl = train_scaled$Survived, 
                  k = 3)

# Build and display the Confusion Matrix [cite: 100, 115]
knn_cm <- confusionMatrix(knn_result, test_scaled$Survived)
print(knn_cm)

# Build the CART model [cite: 769, 780]
cart_model <- rpart(Survived ~ Pclass + Sex + Age + Fare, 
                    data = train_data, 
                    method = "class")

# Plot the tree visually [cite: 11, 954, 966]
rpart.plot(cart_model, main = "Titanic Survival Tree", type = 2)

# Predict on test data [cite: 1028, 1032]
